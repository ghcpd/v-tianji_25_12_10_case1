  export const truncateText = (text: string, maxLength: number): string => {
    if (!text || maxLength <= 0) return ''
    if (text.length <= maxLength) return text
    if (maxLength <= 3) return text.substring(0, maxLength)
    return text.substring(0, maxLength - 3) + '...'
  }

- debouncedSearch: Previously immediately invoked the callback and ignored the delay; now returns a debounced function (using lodash.debounce) so callers can call the returned function and it will debounce correctly.
  Corrected code:
  export const debouncedSearch = (callback: (...args: any[]) => void, delay: number = 300) => {
    return debounce(callback, delay)
  }

- groupBy: Falsy keys like 0 or '' were coerced to 'undefined' due to `||`. Now treats 0 and empty string as valid keys and only treats null/undefined as 'undefined'.
  Corrected code:
  export const groupBy = <T>(array: T[], key: keyof T): Record<string, T[]> => {
    if (!array || array.length === 0) return {}
    return array.reduce((result, item) => {
      const keyVal = item[key]
      const groupKey = String(keyVal === undefined || keyVal === null ? 'undefined' : keyVal)
      if (!result[groupKey]) {
        result[groupKey] = []
      }
      result[groupKey].push(item)
      return result
    }, {} as Record<string, T[]>)
  }
